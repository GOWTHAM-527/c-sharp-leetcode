public class Solution {
    private static readonly string[] Mapping = {
        "",     // 0
        "",     // 1
        "abc",  // 2
        "def",  // 3
        "ghi",  // 4
        "jkl",  // 5
        "mno",  // 6
        "pqrs", // 7
        "tuv",  // 8
        "wxyz"  // 9
    };

    public IList<string> LetterCombinations(string digits) {
        IList<string> result = new List<string>();
        if (string.IsNullOrEmpty(digits)) {
            return result;
        }

        Backtrack(result, digits, new System.Text.StringBuilder(), 0);
        return result;
    }

    private void Backtrack(IList<string> result, string digits, System.Text.StringBuilder current, int index) {
        if (index == digits.Length) {
            result.Add(current.ToString());
            return;
        }

        string letters = Mapping[digits[index] - '0'];
        for (int i = 0; i < letters.Length; i++) {
            current.Append(letters[i]);
            Backtrack(result, digits, current, index + 1);
            current.Length--; 
        }
    }
}
